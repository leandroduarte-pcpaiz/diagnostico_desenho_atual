@dataclass
class Ponto:
    x: float
    y: float


@dataclass
class Segmento:
    indice: int
    x1: float
    y1: float
    x2: float
    y2: float

    @property
    def comprimento(self) -> float:
        return math.hypot(
            self.x2 - self.x1,
            self.y2 - self.y1,
        )

    @property
    def dx(self) -> float:
        return self.x2 - self.x1

    @property
    def dy(self) -> float:
        return self.y2 - self.y1

    @property
    def angulo(self) -> float:
        return math.degrees(
            math.atan2(
                self.dy,
                self.dx,
            )
        )

def distancia(
    p1: Ponto,
    p2: Ponto,
) -> float:

    return math.hypot(
        p2.x - p1.x,
        p2.y - p1.y,
    )


def ponto_proximo(
    p1: Ponto,
    p2: Ponto,
    tolerancia: float = TOLERANCIA_PONTO,
) -> bool:

    return distancia(p1, p2) <= tolerancia


def normalizar_componente(valor) -> str:

    if valor is None:
        return ""

    texto = str(valor).strip()

    numeros = re.findall(
        r"\d+",
        texto,
    )

    if not numeros:
        return texto

    return numeros[-1].zfill(3)


def limpar_texto(
    texto: str,
) -> str:

    if texto is None:
        return ""

    texto = texto.replace(
        "\n",
        " ",
    )

    texto = texto.replace(
        "\r",
        " ",
    )

    return re.sub(
        r"\s+",
        " ",
        texto,
    ).strip()

# DISTÃ‚NCIA PONTO-LINHA
# =============================================================================

def distancia_ponto_segmento(
    ponto: Ponto,
    segmento: Segmento,
) -> float:

    x = ponto.x
    y = ponto.y

    x1 = segmento.x1
    y1 = segmento.y1

    x2 = segmento.x2
    y2 = segmento.y2

    dx = x2 - x1
    dy = y2 - y1

    if abs(dx) < 1e-12 and abs(dy) < 1e-12:
        return math.hypot(
            x - x1,
            y - y1,
        )

    t = (
        (x - x1) * dx +
        (y - y1) * dy
    ) / (
        dx * dx +
        dy * dy
    )

    t = max(
        0.0,
        min(
            1.0,
            t,
        ),
    )

    px = x1 + t * dx
    py = y1 + t * dy

    return math.hypot(
        x - px,
        y - py,
    )

# ÃREA DE POLÃGONO
# =============================================================================

def area_pontos(
    pontos: List[Ponto],
) -> float:

    if len(pontos) < 3:
        return 0.0

    soma = 0.0

    for i in range(len(pontos)):

        p1 = pontos[i]

        p2 = pontos[
            (i + 1) % len(pontos)
        ]

        soma += (
            p1.x * p2.y
            -
            p2.x * p1.y
        )

    return abs(soma) / 2.0


# =============================================================================
# CONTORNO -> PONTOS
# =============================================================================

def segmentos_para_pontos(
    segmentos: List[Segmento],
) -> List[Ponto]:

    if not segmentos:
        return []

    restantes = segmentos.copy()

    primeiro = restantes.pop(0)

    pontos = [
        Ponto(
            primeiro.x1,
            primeiro.y1,
        )
    ]

    atual = Ponto(
        primeiro.x2,
        primeiro.y2,
    )

    pontos.append(atual)

    while restantes:

        encontrado = None
        proximo = None

        for i, seg in enumerate(restantes):

            p_inicio = Ponto(
                seg.x1,
                seg.y1,
            )

            p_fim = Ponto(
                seg.x2,
                seg.y2,
            )

            if ponto_proximo(
                atual,
                p_inicio,
            ):

                encontrado = i
                proximo = p_fim
                break

            if ponto_proximo(
                atual,
                p_fim,
            ):

                encontrado = i
                proximo = p_inicio
                break

        if encontrado is None:
            break

        restantes.pop(encontrado)

        pontos.append(
            proximo
        )

        atual = proximo

    return pontos


# =============================================================================
# BOUNDING BOX
# =============================================================================

def bounding_box_segmentos(
    segmentos: List[Segmento],
) -> Tuple[
    float,
    float,
    float,
    float,
]:

    if not segmentos:
        return (
            0.0,
            0.0,
            0.0,
            0.0,
        )

    xs = []
    ys = []

    for s in segmentos:

        xs.append(s.x1)
        xs.append(s.x2)

        ys.append(s.y1)
        ys.append(s.y2)

    return (
        min(xs),
        min(ys),
        max(xs),
        max(ys),
    )


# =============================================================================
# CONECTIVIDADE
# =============================================================================

def chave_ponto(
    x: float,
    y: float,
    tolerancia: float = TOLERANCIA_PONTO,
):

    return (
        round(x / tolerancia),
        round(y / tolerancia),
    )


def agrupar_segmentos_conectados(
    segmentos: List[Segmento],
) -> List[List[Segmento]]:

    if not segmentos:
        return []

    pontos = defaultdict(list)

    for i, seg in enumerate(segmentos):

        pontos[
            chave_ponto(
                seg.x1,
                seg.y1,
            )
        ].append(i)

        pontos[
            chave_ponto(
                seg.x2,
                seg.y2,
            )
        ].append(i)

    visitados = set()

    grupos = []

    for inicio in range(
        len(segmentos)
    ):

        if inicio in visitados:
            continue

        fila = deque([inicio])

        visitados.add(inicio)

        grupo = []

        while fila:

            atual = fila.popleft()

            grupo.append(
                segmentos[atual]
            )

            seg = segmentos[atual]

            chaves = [
                chave_ponto(
                    seg.x1,
                    seg.y1,
                ),
                chave_ponto(
                    seg.x2,
                    seg.y2,
                ),
            ]

            vizinhos = set()

            for chave in chaves:

                for idx in pontos[chave]:
                    vizinhos.add(idx)

            for idx in vizinhos:

                if idx not in visitados:

                    visitados.add(idx)

                    fila.append(idx)

        grupos.append(grupo)

    return grupos


# =============================================================================
# CONTORNO FECHADO
# =============================================================================

def contorno_fechado(
    segmentos: List[Segmento],
    tolerancia: float = TOLERANCIA_PONTO,
) -> bool:

    if len(segmentos) < 3:
        return False

    extremos = []

    for s in segmentos:

        extremos.append(
            Ponto(
                s.x1,
                s.y1,
            )
        )

        extremos.append(
            Ponto(
                s.x2,
                s.y2,
            )
        )

    grupos = []

    for p in extremos:

        encontrado = False

        for grupo in grupos:

            if ponto_proximo(
                p,
                grupo[0],
                tolerancia,
            ):

                grupo.append(p)

                encontrado = True
                break

        if not encontrado:
            grupos.append([p])

    for grupo in grupos:

        if len(grupo) < 2:
            return False

    return True


# =============================================================================
# CRIAR CONTORNO
# =============================================================================

def criar_contorno(
    segmentos: List[Segmento],
    componente: str,
) -> Optional[Contorno]:

    if len(segmentos) < 3:
        return None

    if not contorno_fechado(
        segmentos
    ):
        return None

    xmin, ymin, xmax, ymax = (
        bounding_box_segmentos(
            segmentos
        )
    )

    largura = xmax - xmin
    altura = ymax - ymin

    if (
        largura < MIN_ALTURA_PECA
        or
        altura < MIN_ALTURA_PECA
    ):
        return None

    pontos = segmentos_para_pontos(
        segmentos
    )

    if len(pontos) < 3:
        return None

    area = area_pontos(
        pontos
    )

    if area < MIN_AREA_CONTORNO:
        return None

    perimetro = sum(
        s.comprimento
        for s in segmentos
    )

    return Contorno(
        componente=normalizar_componente(
            componente
        ),
        segmentos=segmentos,
        largura_pdf=largura,
        altura_pdf=altura,
        area_pdf=area,
        perimetro_pdf=perimetro,
    )


# =============================================================================
# ESCALA
# =============================================================================

def calcular_escala(
    contorno: Contorno,
    alvo_largura: float,
    alvo_altura: float,
) -> Tuple[
    float,
    float,
    float,
]:

    largura = contorno.largura_pdf
    altura = contorno.altura_pdf

    if largura <= 0 or altura <= 0:

        return (
            0.0,
            999999.0,
            999999.0,
        )

    erro_normal = (
        abs(
            largura - alvo_largura
        )
        /
        alvo_largura
        +
        abs(
            altura - alvo_altura
        )
        /
        alvo_altura
    )

    erro_rotacionado = (
        abs(
            largura - alvo_altura
        )
        /
        alvo_altura
        +
        abs(
            altura - alvo_largura
        )
        /
        alvo_largura
    )

    if erro_rotacionado < erro_normal:

        escala = (
            alvo_largura / altura
            +
            alvo_altura / largura
        ) / 2.0

        estimada_largura = (
            altura * escala
        )

        estimada_altura = (
            largura * escala
        )

    else:

        escala = (
            alvo_largura / largura
            +
            alvo_altura / altura
        ) / 2.0

        estimada_largura = (
            largura * escala
        )

        estimada_altura = (
            altura * escala
        )

    erro_dimensao = (
        abs(
            estimada_largura -
            alvo_largura
        )
        /
        alvo_largura
        +
        abs(
            estimada_altura -
            alvo_altura
        )
        /
        alvo_altura
    ) / 2.0

    proporcao_pdf = (
        largura / altura
    )

    proporcao_alvo = (
        alvo_largura / alvo_altura
    )

    erro_proporcao_pdf = (
        abs(
            proporcao_pdf -
            proporcao_alvo
        )
        /
        proporcao_alvo
    )

    return (
        escala,
        erro_proporcao_pdf,
        erro_dimensao,
    )


# =============================================================================
# SCORE
# =============================================================================

def pontuar_contorno(
    contorno: Contorno,
    alvo_largura: float,
    alvo_altura: float,
) -> Contorno:

    escala, erro_prop, erro_dim = (
        calcular_escala(
            contorno,
            alvo_largura,
            alvo_altura,
        )
    )

    score = 0.0

    if erro_prop <= 0.01:
        score += 50

    elif erro_prop <= 0.03:
        score += 30

    elif erro_prop <= 0.10:
        score += 10

    else:
        score -= 40

    if erro_dim <= 0.01:
        score += 50

    elif erro_dim <= 0.03:
        score += 30

    elif erro_dim <= 0.10:
        score += 10

    else:
        score -= 40

    if len(contorno.segmentos) >= 40:
        score += 20

    elif len(contorno.segmentos) >= 20:
        score += 10

    relacao = (
        max(
            contorno.largura_pdf,
            contorno.altura_pdf,
        )
        /
        max(
            min(
                contorno.largura_pdf,
                contorno.altura_pdf,
            ),
            0.0001,
        )
    )

    if relacao >= 3:
        score += 10

    contorno.score = score
    contorno.escala = escala

    largura_est = (
        contorno.largura_pdf *
        escala
    )

    altura_est = (
        contorno.altura_pdf *
        escala
    )

    erro_normal = (
        abs(
            largura_est -
            alvo_largura
        )
        +
        abs(
            altura_est -
            alvo_altura
        )
    )

    largura_rot = (
        contorno.altura_pdf *
        escala
    )

    altura_rot = (
        contorno.largura_pdf *
        escala
    )

    erro_rot = (
        abs(
            largura_rot -
            alvo_largura
        )
        +
        abs(
            altura_rot -
            alvo_altura
        )
    )

    if erro_rot < erro_normal:

        contorno.largura_mm = largura_rot
        contorno.altura_mm = altura_rot

    else:

        contorno.largura_mm = largura_est
        contorno.altura_mm = altura_est

    contorno.area_mm2 = (
        contorno.area_pdf *
        escala *
        escala
    )

    contorno.perimetro_mm = (
        contorno.perimetro_pdf *
        escala
    )

    return contorno

# EXTRAÃ‡ÃƒO DE LINHAS
# =============================================================================

def extrair_linhas(
    pagina,
) -> List[Segmento]:

    segmentos = []

    desenhos = pagina.get_drawings()

    contador = 0

    for item in desenhos:

        for elemento in item.get(
            "items",
            [],
        ):

            if not elemento:
                continue

            tipo = elemento[0]

            if tipo != "l":
                continue

            p1 = elemento[1]
            p2 = elemento[2]

            x1 = float(p1.x)
            y1 = float(p1.y)

            x2 = float(p2.x)
            y2 = float(p2.y)

            comprimento = math.hypot(
                x2 - x1,
                y2 - y1,
            )

            if comprimento < MIN_SEGMENTO:
                continue

            segmentos.append(
                Segmento(
                    indice=contador,
                    x1=x1,
                    y1=y1,
                    x2=x2,
                    y2=y2,
                )
            )

            contador += 1

    return segmentos

# EXTRAÃ‡ÃƒO DE TEXTOS
# =============================================================================

def extrair_textos(
    pagina,
) -> List[dict]:

    resultado = []

    dados = pagina.get_text(
        "dict"
    )

    for bloco in dados.get(
        "blocks",
        [],
    ):

        if "lines" not in bloco:
            continue

        for linha in bloco["lines"]:

            for span in linha.get(
                "spans",
                [],
            ):

                texto = limpar_texto(
                    span.get(
                        "text",
                        "",
                    )
                )

                if not texto:
                    continue

                bbox = span.get(
                    "bbox",
                    None,
                )

                resultado.append(
                    {
                        "texto": texto,
                        "x0": (
                            bbox[0]
                            if bbox
                            else 0
                        ),
                        "y0": (
                            bbox[1]
                            if bbox
                            else 0
                        ),
                        "x1": (
                            bbox[2]
                            if bbox
                            else 0
                        ),
                        "y1": (
                            bbox[3]
                            if bbox
                            else 0
                        ),
                    }
                )

    return resultado

# EXTRAIR NÃšMEROS
# =============================================================================

def extrair_numeros_textuais(
    textos: List[dict],
) -> List[float]:

    valores = []

    padroes = [
        r"(?<!\d)(\d+(?:[.,]\d+)?)\s*mm\b",
        r"(?<!\d)(\d{2,5}(?:[.,]\d+)?)",
    ]

    for item in textos:

        texto = item["texto"]

        for padrao in padroes:

            encontrados = re.findall(
                padrao,
                texto,
                flags=re.IGNORECASE,
            )

            for valor in encontrados:

                valor = valor.replace(
                    ",",
                    ".",
                )

                try:

                    numero = float(valor)

                    if 1 <= numero <= 10000:
                        valores.append(numero)

                except ValueError:
                    pass

    return valores


# =============================================================================
# IDENTIFICAR ESPESSURA
# =============================================================================

def identificar_espessura(
    textos: List[dict],
) -> Optional[float]:

    padroes = [
        r"(?:ESP|ESPESSURA|THK|T)\s*[:=]?\s*(\d+(?:[.,]\d+)?)",
        r"(\d+(?:[.,]\d+)?)\s*MM\b",
    ]

    candidatos = []

    for item in textos:

        texto = item["texto"].upper()

        for padrao in padroes:

            encontrados = re.findall(
                padrao,
                texto,
            )

            for valor in encontrados:

                try:

                    numero = float(
                        valor.replace(
                            ",",
                            ".",
                        )
                    )

                    if 0.5 <= numero <= 30:
                        candidatos.append(
                            numero
                        )

                except ValueError:
                    pass

    if not candidatos:
        return None

    candidatos.sort(
        key=lambda x: abs(
            x - ESPESSURA_PADRAO
        )
    )

    return candidatos[0]

def extrair_raios(

        for padrao in padroes:

            valores = re.findall(
                padrao,
                texto,
            )

            for valor in valores:

                try:

                    raio = float(
                        valor.replace(
                            ",",
                            ".",
                        )
                    )

                    if 0.5 <= raio <= 100:
                        encontrados.append(
                            raio
                        )

                except ValueError:
                    pass

    return encontrados

# FUNÃ‡Ã•ES GEOMÃ‰TRICAS PARA DOBRAS
# =============================================================================

def angulo_diferenca(
    a: float,
    b: float,
) -> float:

    a = abs(a) % 180.0
    b = abs(b) % 180.0

    diferenca = abs(
        a - b
    )

    return min(
        diferenca,
        180.0 - diferenca,
    )


def eixo_longitudinal(
    contorno: Contorno,
) -> str:
    """
    Retorna o eixo LONGITUDINAL NO PDF.

    IMPORTANTE:
    Esta funcao trabalha exclusivamente com a orientacao
    geometrica original do desenho.

    A geometria do PDF pode estar rotacionada em relacao
    a orientacao fisica da peca.

    Portanto:

        largura_pdf / altura_pdf
            -> orientacao usada para detectar as linhas de dobra

        largura_mm / altura_mm
            -> orientacao fisica usada posteriormente no
               desenvolvimento e no blank.

    Nao misturar os dois sistemas.
    """

    if (
        contorno.largura_pdf
        >=
        contorno.altura_pdf
    ):
        return "X"

    return "Y"


def coordenada_transversal(
    segmento: Segmento,
    eixo_longitudinal: str,
) -> float:

    if eixo_longitudinal == "X":

        return (
            segmento.y1 +
            segmento.y2
        ) / 2.0

    return (
        segmento.x1 +
        segmento.x2
    ) / 2.0


def coordenada_longitudinal(
    segmento: Segmento,
    eixo_longitudinal: str,
) -> Tuple[float, float]:

    if eixo_longitudinal == "X":

        return (
            min(
                segmento.x1,
                segmento.x2,
            ),
            max(
                segmento.x1,
                segmento.x2,
            ),
        )

    return (
        min(
            segmento.y1,
            segmento.y2,
        ),
        max(
            segmento.y1,
            segmento.y2,
        ),
    )


# =============================================================================
# NOVA FUNÃ‡ÃƒO:
# IDENTIFICAR SEGMENTOS LONGITUDINAIS DO CONTORNO
# =============================================================================

def identificar_segmentos_longitudinais(
    contorno: Contorno,
) -> dict:

    eixo = eixo_longitudinal(
        contorno
    )

    escala = (
        contorno.escala
        if contorno.escala > 0
        else 1.0
    )

    xmin, ymin, xmax, ymax = (
        bounding_box_segmentos(
            contorno.segmentos
        )
    )

    if eixo == "X":

        comprimento_total_pdf = (
            xmax - xmin
        )

        comprimento_total_mm = (
            comprimento_total_pdf *
            escala
        )

        angulo_referencia = 0.0

    else:

        comprimento_total_pdf = (
            ymax - ymin
        )

        comprimento_total_mm = (
            comprimento_total_pdf *
            escala
        )

        angulo_referencia = 90.0

    candidatos = []

    if comprimento_total_pdf <= 0:
        return {
            "eixo_longitudinal": eixo,
            "comprimento_total_mm": 0.0,
            "candidatos": [],
            "trilhos": [],
        }

    for segmento in contorno.segmentos:

        comprimento_pdf = (
            segmento.comprimento
        )

        if comprimento_pdf <= 0:
            continue

        fracao = (
            comprimento_pdf /
            comprimento_total_pdf
        )

        if (
            fracao <
            MIN_FRACAO_SEGMENTO_LONGITUDINAL
        ):
            continue

        diferenca = angulo_diferenca(
            segmento.angulo,
            angulo_referencia,
        )

        if diferenca > TOLERANCIA_ANGULO_DOBRA:
            continue

        coordenada = coordenada_transversal(
            segmento,
            eixo,
        )

        inicio, fim = coordenada_longitudinal(
            segmento,
            eixo,
        )

        candidatos.append(
            {
                "segmento_indice":
                    segmento.indice,

                "coordenada_transversal_pdf":
                    coordenada,

                "coordenada_transversal_mm":
                    (
                        coordenada -
                        (
                            ymin
                            if eixo == "X"
                            else xmin
                        )
                    ) * escala,

                "inicio_longitudinal_pdf":
                    inicio,

                "fim_longitudinal_pdf":
                    fim,

                "comprimento_pdf":
                    comprimento_pdf,

                "comprimento_mm":
                    comprimento_pdf * escala,

                "fracao_comprimento":
                    fracao,

                "angulo_graus":
                    segmento.angulo,

                "diferenca_angular":
                    diferenca,
            }
        )

    candidatos.sort(
        key=lambda item:
            item[
                "coordenada_transversal_mm"
            ]
    )

    # -------------------------------------------------------------------------
    # AGRUPAMENTO DOS TRILHOS
    # -------------------------------------------------------------------------

    trilhos = []

    for candidato in candidatos:

        coord = candidato[
            "coordenada_transversal_mm"
        ]

        grupo = None

        for trilho in trilhos:

            if abs(
                coord -
                trilho[
                    "coordenada_media_mm"
                ]
            ) <= TOLERANCIA_TRILHO_TRANSVERSAL_MM:

                grupo = trilho
                break

        if grupo is None:

            grupo = {
                "coordenada_media_mm":
                    coord,

                "segmentos":
                    [],

                "comprimento_total_mm":
                    0.0,
            }

            trilhos.append(
                grupo
            )

        grupo["segmentos"].append(
            candidato
        )

        grupo["comprimento_total_mm"] += (
            candidato["comprimento_mm"]
        )

        grupo["coordenada_media_mm"] = (
            sum(
                item[
                    "coordenada_transversal_mm"
                ]
                for item in grupo["segmentos"]
            )
            /
            len(
                grupo["segmentos"]
            )
        )

    # -------------------------------------------------------------------------
    # CLASSIFICAÃ‡ÃƒO DOS TRILHOS
    # -------------------------------------------------------------------------

    for numero, trilho in enumerate(
        trilhos,
        start=1,
    ):

        trilho["numero"] = numero

        trilho["quantidade_segmentos"] = len(
            trilho["segmentos"]
        )

        trilho["comprimento_maximo_mm"] = max(
            (
                item["comprimento_mm"]
                for item in trilho["segmentos"]
            ),
            default=0.0,
        )

    return {
        "eixo_longitudinal":
            eixo,

        "comprimento_total_mm":
            comprimento_total_mm,

        "candidatos":
            candidatos,

        "trilhos":
            trilhos,
    }


# =============================================================================
# RECONSTRUIR PERFIL TRANSVERSAL
# =============================================================================

def reconstruir_perfil_transversal(
    contorno: Contorno,
    dobras: List[Dobra],
    espessura_mm: Optional[float],
) -> dict:

    """
    ReconstrÃ³i as trÃªs regiÃµes retas da peÃ§a a partir da geometria
    longitudinal real do contorno.

    A funÃ§Ã£o NÃƒO utiliza:

        1/3
        2/3
        39%
        71%

    Ela procura as arestas longitudinais reais.

    O objetivo Ã© produzir evidÃªncia geomÃ©trica para:

        ABA 1
        DOBRA 1
        ABA 2
        DOBRA 2
        ABA 3

    As posiÃ§Ãµes das dobras continuam sendo obtidas das linhas internas.

    A geometria longitudinal serve para validar que existem trÃªs
    regiÃµes fÃ­sicas da peÃ§a e para fornecer a separaÃ§Ã£o entre
    as faces da chapa.
    """

    if espessura_mm is None:
        espessura_mm = ESPESSURA_PADRAO

    trilhos_info = identificar_segmentos_longitudinais(
        contorno
    )

    trilhos = trilhos_info[
        "trilhos"
    ]

    if len(trilhos) < 3:

        return {
            "status":
                "GEOMETRIA_INSUFICIENTE",

            "motivo":
                "MENOS_DE_TRES_TRILHOS_LONGITUDINAIS",

            "trilhos":
                trilhos,

            "quantidade_trilhos":
                len(trilhos),

            "abas_identificadas":
                [],
        }

    # -------------------------------------------------------------------------
    # Cada flange de uma chapa dobrada aparece como duas faces.
    #
    # Procuramos pares de trilhos separados aproximadamente pela espessura.
    # -------------------------------------------------------------------------

    pares_espessura = []

    for i in range(
        len(trilhos)
    ):

        for j in range(
            i + 1,
            len(trilhos)
        ):

            t1 = trilhos[i]
            t2 = trilhos[j]

            distancia_mm = abs(
                t2[
                    "coordenada_media_mm"
                ]
                -
                t1[
                    "coordenada_media_mm"
                ]
            )

            erro_espessura = abs(
                distancia_mm -
                espessura_mm
            )

            if (
                erro_espessura
                <=
                TOLERANCIA_ESPESSURA_MM
            ):

                pares_espessura.append(
                    {
                        "trilho_1":
                            t1["numero"],

                        "trilho_2":
                            t2["numero"],

                        "distancia_mm":
                            distancia_mm,

                        "erro_espessura_mm":
                            erro_espessura,

                        "score":
                            (
                                t1[
                                    "comprimento_maximo_mm"
                                ]
                                +
                                t2[
                                    "comprimento_maximo_mm"
                                ]
                                -
                                erro_espessura * 10.0
                            ),
                    }
                )

    pares_espessura.sort(
        key=lambda item:
            item["score"],
        reverse=True,
    )

    # -------------------------------------------------------------------------
    # POSIÃ‡Ã•ES DAS DOBRAS
    # -------------------------------------------------------------------------

    posicoes = sorted(
        [
            d.posicao_mm
            for d in dobras
            if d.posicao_mm is not None
        ]
    )

    if len(posicoes) != 2:

        return {
            "status":
                "GEOMETRIA_INSUFICIENTE",

            "motivo":
                "DUAS_POSICOES_DE_DOBRA_NAO_DISPONIVEIS",

            "trilhos":
                trilhos,

            "pares_espessura":
                pares_espessura,

            "abas_identificadas":
                [],
        }

    p1 = posicoes[0]
    p2 = posicoes[1]

    dimensao_transversal = min(
        contorno.largura_mm,
        contorno.altura_mm,
    )

    # -------------------------------------------------------------------------
    # REGIÃ•ES GEOMÃ‰TRICAS EXTERNAS
    # -------------------------------------------------------------------------

    regiao_1 = p1
    regiao_2 = p2 - p1
    regiao_3 = (
        dimensao_transversal -
        p2
    )

    if (
        regiao_1 <= 0
        or regiao_2 <= 0
        or regiao_3 <= 0
    ):

        return {
            "status":
                "GEOMETRIA_INCONSISTENTE",

            "motivo":
                "REGIOES_TRANSVERSAIS_INVALIDAS",

            "trilhos":
                trilhos,

            "pares_espessura":
                pares_espessura,

            "regioes_externas_mm":
                [
                    regiao_1,
                    regiao_2,
                    regiao_3,
                ],

            "abas_identificadas":
                [],
        }

    # -------------------------------------------------------------------------
    # IDENTIFICAÃ‡ÃƒO DAS ABAS
    # -------------------------------------------------------------------------

    abas = [
        {
            "numero":
                1,

            "trecho_externo_mm":
                regiao_1,

            "posicao_inicio_mm":
                0.0,

            "posicao_fim_mm":
                p1,

            "origem":
                "GEOMETRIA_POSICAO_DOBRA_1",
        },

        {
            "numero":
                2,

            "trecho_externo_mm":
                regiao_2,

            "posicao_inicio_mm":
                p1,

            "posicao_fim_mm":
                p2,

            "origem":
                "GEOMETRIA_ENTRE_DOBRAS",
        },

        {
            "numero":
                3,

            "trecho_externo_mm":
                regiao_3,

            "posicao_inicio_mm":
                p2,

            "posicao_fim_mm":
                dimensao_transversal,

            "origem":
                "GEOMETRIA_DOBRA_2_ATE_BORDA",
        },
    ]

    # -------------------------------------------------------------------------
    # VALIDAÃ‡ÃƒO DOS TRILHOS
    # -------------------------------------------------------------------------

    comprimento_longitudinal = (
        trilhos_info[
            "comprimento_total_mm"
        ]
    )

    trilhos_relevantes = [
        t
        for t in trilhos
        if (
            t[
                "comprimento_maximo_mm"
            ]
            >=
            comprimento_longitudinal * 0.70
        )
    ]

    geometria_confirmada = (
        len(trilhos_relevantes) >= 3
    )

    return {
        "status":
            (
                "CONFIRMADO"
                if geometria_confirmada
                else
                "PARCIAL"
            ),

        "motivo":
            (
                "TRES_REGIOES_LONGITUDINAIS_IDENTIFICADAS"
                if geometria_confirmada
                else
                "REGIOES_DEFINIDAS_PELAS_DOBRAS_MAS_TRILHOS_INSUFICIENTES"
            ),

        "eixo_longitudinal":
            trilhos_info[
                "eixo_longitudinal"
            ],

        "comprimento_longitudinal_mm":
            comprimento_longitudinal,

        "dimensao_transversal_mm":
            dimensao_transversal,

        "trilhos":
            trilhos,

        "trilhos_relevantes":
            trilhos_relevantes,

        "pares_espessura":
            pares_espessura,

        "regioes_externas_mm":
            [
                regiao_1,
                regiao_2,
                regiao_3,
            ],

        "abas_identificadas":
            abas,
    }


# =============================================================================
# DETECTAR LINHAS DE DOBRA
# =============================================================================

def detectar_linhas_dobra_geometricas(
    segmentos: List[Segmento],
    contorno: Contorno,
    espessura_mm: Optional[float],
    raio_mm: Optional[float],
) -> dict:

    eixo = eixo_longitudinal(
        contorno
    )

    xmin, ymin, xmax, ymax = (
        bounding_box_segmentos(
            contorno.segmentos
        )
    )

    if eixo == "X":

        comprimento_longitudinal_pdf = (
            xmax - xmin
        )

        limite_transversal_min = ymin
        limite_transversal_max = ymax

    else:

        comprimento_longitudinal_pdf = (
            ymax - ymin
        )

        limite_transversal_min = xmin
        limite_transversal_max = xmax

    if comprimento_longitudinal_pdf <= 0:

        return {
            "encontradas": False,
            "quantidade": 0,
            "eixo_longitudinal": eixo,
            "eixo_desenvolvimento": (
                "Y" if eixo == "X" else "X"
            ),
            "candidatos": [],
            "linhas_selecionadas": [],
        }

    escala = contorno.escala

    if escala <= 0:
        escala = 1.0

    comprimento_longitudinal_mm = (
        comprimento_longitudinal_pdf *
        escala
    )

    distancia_borda_pdf = (
        MIN_DISTANCIA_BORDA_DOBRA_MM /
        escala
    )

    candidatos_brutos = []

    for segmento in segmentos:

        comprimento_pdf = (
            segmento.comprimento
        )

        if comprimento_pdf <= 0:
            continue

        comprimento_mm = (
            comprimento_pdf *
            escala
        )

        fracao = (
            comprimento_pdf /
            comprimento_longitudinal_pdf
        )

        if (
            fracao <
            MIN_FRACAO_COMPRIMENTO_LINHA_DOBRA
        ):
            continue

        angulo = segmento.angulo

        if eixo == "X":

            diferenca_angular = angulo_diferenca(
                angulo,
                0.0,
            )

        else:

            diferenca_angular = angulo_diferenca(
                angulo,
                90.0,
            )

        if (
            diferenca_angular >
            TOLERANCIA_ANGULO_DOBRA
        ):
            continue

        coord_transversal = (
            coordenada_transversal(
                segmento,
                eixo,
            )
        )

        if (
            coord_transversal <
            limite_transversal_min
        ):
            continue

        if (
            coord_transversal >
            limite_transversal_max
        ):
            continue

        distancia_borda = min(
            abs(
                coord_transversal -
                limite_transversal_min
            ),
            abs(
                limite_transversal_max -
                coord_transversal
            ),
        )

        if (
            distancia_borda <
            distancia_borda_pdf
        ):
            continue

        xlong1, xlong2 = (
            coordenada_longitudinal(
                segmento,
                eixo,
            )
        )

        # A linha de dobra deve estar efetivamente dentro do comprimento
        # longitudinal do contorno principal. Isso evita que cotas,
        # linhas de centro ou outras geometrias do desenho sejam
        # confundidas com linhas de dobra.
        if eixo == "X":
            limite_longitudinal_min = xmin
            limite_longitudinal_max = xmax
        else:
            limite_longitudinal_min = ymin
            limite_longitudinal_max = ymax

        sobreposicao = max(
            0.0,
            min(xlong2, limite_longitudinal_max)
            -
            max(xlong1, limite_longitudinal_min),
        )

        comprimento_interno = xlong2 - xlong1

        if comprimento_interno <= 0:
            continue

        # A linha precisa cobrir pelo menos 70% do comprimento do
        # contorno. O teste de fração acima já limita o comprimento
        # absoluto; este segundo teste exige que ela também esteja
        # realmente sobre o contorno selecionado.
        fracao_sobreposicao = (
            sobreposicao / comprimento_longitudinal_pdf
        )

        if fracao_sobreposicao < 0.70:
            continue

        score_comprimento = min(
            fracao,
            1.0,
        ) * 60.0

        score_angulo = max(
            0.0,
            20.0 -
            diferenca_angular * 10.0,
        )

        score_borda = min(
            distancia_borda *
            escala /
            20.0,
            10.0,
        )

        score = (
            score_comprimento
            +
            score_angulo
            +
            score_borda
        )

        candidatos_brutos.append(
            {
                "segmento_indice":
                    segmento.indice,

                "coordenada_transversal_pdf":
                    coord_transversal,

                "comprimento_pdf":
                    comprimento_pdf,

                "comprimento_mm":
                    comprimento_mm,

                "fracao_comprimento":
                    fracao,

                "angulo_graus":
                    angulo,

                "diferenca_angular":
                    diferenca_angular,

                "score":
                    score,
            }
        )

    candidatos_brutos.sort(
        key=lambda item:
            item[
                "coordenada_transversal_pdf"
            ]
    )

    grupos = []

    tolerancia_agrupamento_pdf = (
        TOLERANCIA_AGRUPAMENTO_DOBRA_MM /
        escala
    )

    for candidato in candidatos_brutos:

        coord = candidato[
            "coordenada_transversal_pdf"
        ]

        grupo_encontrado = None

        for grupo in grupos:

            media = grupo[
                "media_coordenada_pdf"
            ]

            if abs(
                coord - media
            ) <= tolerancia_agrupamento_pdf:

                grupo["itens"].append(
                    candidato
                )

                total = len(
                    grupo["itens"]
                )

                grupo[
                    "media_coordenada_pdf"
                ] = (
                    sum(
                        item[
                            "coordenada_transversal_pdf"
                        ]
                        for item in grupo["itens"]
                    )
                    /
                    total
                )

                grupo_encontrado = grupo
                break

        if grupo_encontrado is None:

            grupos.append(
                {
                    "media_coordenada_pdf":
                        coord,

                    "itens":
                        [candidato],
                }
            )

    candidatos_agrupados = []

    for grupo in grupos:

        itens = grupo["itens"]

        comprimento_total_pdf = sum(
            item["comprimento_pdf"]
            for item in itens
        )

        comprimento_total_mm = (
            comprimento_total_pdf *
            escala
        )

        media_coord = grupo[
            "media_coordenada_pdf"
        ]

        melhor_score = max(
            item["score"]
            for item in itens
        )

        candidatos_agrupados.append(
            {
                "coordenada_transversal_pdf":
                    media_coord,

                "posicao_mm_desde_borda":
                    (
                        media_coord -
                        limite_transversal_min
                    ) *
                    escala,

                "comprimento_total_pdf":
                    comprimento_total_pdf,

                "comprimento_total_mm":
                    comprimento_total_mm,

                "quantidade_segmentos":
                    len(itens),

                "score":
                    melhor_score,

                "segmentos":
                    itens,
            }
        )

    candidatos_uteis = []

    for candidato in candidatos_agrupados:

        posicao = candidato[
            "posicao_mm_desde_borda"
        ]

        dimensao_transversal_mm = (
            limite_transversal_max -
            limite_transversal_min
        ) * escala

        distancia_borda_1 = posicao

        distancia_borda_2 = (
            dimensao_transversal_mm -
            posicao
        )

        if (
            distancia_borda_1 <
            MIN_DISTANCIA_BORDA_DOBRA_MM
        ):
            continue

        if (
            distancia_borda_2 <
            MIN_DISTANCIA_BORDA_DOBRA_MM
        ):
            continue

        candidatos_uteis.append(
            candidato
        )

    candidatos_uteis.sort(
        key=lambda item:
            item[
                "posicao_mm_desde_borda"
            ]
    )

    par_selecionado = None
    score_par = -999999.0

    raio = (
        raio_mm
        if raio_mm is not None
        else RAIO_PADRAO
    )

    espessura = (
        espessura_mm
        if espessura_mm is not None
        else ESPESSURA_PADRAO
    )

    setback_referencia = (
        (
            raio +
            espessura
        )
        *
        math.tan(
            math.radians(
                ANGULO_PADRAO / 2.0
            )
        )
    )

    dimensao_transversal_mm = (
        limite_transversal_max -
        limite_transversal_min
    ) * escala

    for i in range(
        len(candidatos_uteis)
    ):

        for j in range(
            i + 1,
            len(candidatos_uteis)
        ):

            c1 = candidatos_uteis[i]
            c2 = candidatos_uteis[j]

            p1 = c1[
                "posicao_mm_desde_borda"
            ]

            p2 = c2[
                "posicao_mm_desde_borda"
            ]

            trecho1 = p1
            trecho2 = p2 - p1
            trecho3 = (
                dimensao_transversal_mm -
                p2
            )

            if trecho1 <= 0:
                continue

            if trecho2 <= 0:
                continue

            if trecho3 <= 0:
                continue

            if trecho1 <= setback_referencia:
                continue

            if trecho2 <= 2.0 * setback_referencia:
                continue

            if trecho3 <= setback_referencia:
                continue

            score_atual = (
                c1["score"]
                +
                c2["score"]
            )

            distancia_entre = (
                p2 - p1
            )

            if (
                distancia_entre
                <
                3.0 * setback_referencia
            ):
                score_atual -= 20.0

            if score_atual > score_par:

                score_par = score_atual

                par_selecionado = (
                    c1,
                    c2,
                )

    encontradas = (
        par_selecionado is not None
    )

    linhas_selecionadas = []

    if encontradas:

        for numero, candidato in enumerate(
            par_selecionado,
            start=1,
        ):

            linhas_selecionadas.append(
                {
                    "numero":
                        numero,

                    "posicao_mm":
                        candidato[
                            "posicao_mm_desde_borda"
                        ],

                    "coordenada_pdf":
                        candidato[
                            "coordenada_transversal_pdf"
                        ],

                    "comprimento_mm":
                        candidato[
                            "comprimento_total_mm"
                        ],

                    "score":
                        candidato[
                            "score"
                        ],

                    "segmentos":
                        candidato[
                            "segmentos"
                        ],
                }
            )

    return {
        "encontradas":
            encontradas,

        "quantidade":
            len(linhas_selecionadas),

        "eixo_longitudinal":
            eixo,

        "eixo_desenvolvimento":
            (
                "TRANSVERSAL"
                if eixo == "X"
                else "LONGITUDINAL"
            ),

        "comprimento_longitudinal_mm":
            comprimento_longitudinal_mm,

        "dimensao_transversal_mm":
            dimensao_transversal_mm,

        "candidatos":
            candidatos_uteis,

        "linhas_selecionadas":
            linhas_selecionadas,

        "score_par":
            (
                score_par
                if encontradas
                else None
            ),

        "setback_referencia_mm":
            setback_referencia,

        "motivo":
            (
                "DUAS_LINHAS_INTERNAS_COMPATIVEIS"
                if encontradas
                else
                "LINHAS_DE_DOBRA_NAO_IDENTIFICADAS"
            ),
    }


# =============================================================================
# IDENTIFICAR DOBRAS
# =============================================================================

def identificar_dobras(
    textos: List[dict],
    contorno: Contorno,
    espessura_mm: Optional[float],
    todos_segmentos: List[Segmento],
) -> Tuple[
    List[Dobra],
    dict,
]:

    angulos_texto = extrair_angulos(
        textos
    )

    raios_texto = extrair_raios(
        textos
    )

    if len(angulos_texto) >= 2:

        angulos = angulos_texto[:2]
        origem_angulo = "TEXTO"

    elif len(angulos_texto) == 1:

        angulos = [
            angulos_texto[0],
            angulos_texto[0],
        ]

        origem_angulo = "TEXTO_E_REPETICAO"

    else:

        angulos = [
            ANGULO_PADRAO,
            ANGULO_PADRAO,
        ]

        origem_angulo = "CONFIGURACAO_PADRAO"

    if len(raios_texto) >= 2:

        raios = raios_texto[:2]
        origem_raio = "TEXTO"

    elif len(raios_texto) == 1:

        raios = [
            raios_texto[0],
            raios_texto[0],
        ]

        origem_raio = "TEXTO_E_REPETICAO"

    else:

        raios = [
            RAIO_PADRAO,
            RAIO_PADRAO,
        ]

        origem_raio = "CONFIGURACAO_PADRAO"

    raio_referencia = raios[0]

    deteccao_geometrica = (
        detectar_linhas_dobra_geometricas(
            todos_segmentos,
            contorno,
            espessura_mm,
            raio_referencia,
        )
    )

    dobras = []

    if (
        deteccao_geometrica["encontradas"]
        and
        deteccao_geometrica["quantidade"] >= 2
    ):

        linhas = (
            deteccao_geometrica[
                "linhas_selecionadas"
            ]
        )

        for i in range(2):

            angulo = (
                angulos[i]
                if i < len(angulos)
                else ANGULO_PADRAO
            )

            raio = (
                raios[i]
                if i < len(raios)
                else RAIO_PADRAO
            )

            linha = linhas[i]

            confianca = 0.97

            if (
                origem_angulo ==
                "CONFIGURACAO_PADRAO"
            ):
                confianca -= 0.03

            if (
                origem_raio ==
                "CONFIGURACAO_PADRAO"
            ):
                confianca -= 0.03

            dobras.append(
                Dobra(
                    numero=i + 1,
                    angulo=angulo,
                    raio=raio,
                    posicao_mm=(
                        linha[
                            "posicao_mm"
                        ]
                    ),
                    direcao=(
                        "PARALELA_EIXO_LONGITUDINAL"
                    ),
                    confianca=confianca,
                    origem=(
                        "GEOMETRIA_LINHA_INTERNA"
                    ),
                )
            )

    else:

        for i in range(
            QUANTIDADE_DOBRAS_PADRAO
        ):

            angulo = (
                angulos[i]
                if i < len(angulos)
                else ANGULO_PADRAO
            )

            raio = (
                raios[i]
                if i < len(raios)
                else RAIO_PADRAO
            )

            dobras.append(
                Dobra(
                    numero=i + 1,
                    angulo=angulo,
                    raio=raio,
                    posicao_mm=None,
                    direcao=None,
                    confianca=0.30,
                    origem=(
                        "DOBRA_IDENTIFICADA_SEM_POSICAO_GEOMETRICA"
                    ),
                )
            )

    return (
        dobras,
        deteccao_geometrica,
    )


# =============================================================================
# ORDENAR DOBRAS
# =============================================================================

def ordenar_dobras(
    dobras: List[Dobra],
) -> List[Dobra]:

    if not dobras:
        return []

    return sorted(
        dobras,
        key=lambda d: (
            d.posicao_mm
            if d.posicao_mm is not None
            else 999999
        ),
    )


# =============================================================================
# FATOR K
# =============================================================================

def fator_k_por_raio(
    raio: Optional[float],
    espessura: Optional[float],
) -> float:

    if espessura is None:
        return 0.33

    if raio is None:
        return 0.33

    relacao = (
        raio /
        espessura
    )

    if relacao <= 1.0:
        return 0.33

    if relacao <= 1.5:
        return 0.35

    if relacao <= 2.0:
        return 0.38

    if relacao <= 3.0:
        return 0.40

    return 0.42


# =============================================================================
# BEND ALLOWANCE
# =============================================================================

def bend_allowance(
    angulo: float,
    raio: float,
    espessura: float,
    k: float,
) -> float:

    theta = math.radians(
        abs(angulo)
    )

    return (
        theta *
        (
            raio +
            k * espessura
        )
    )


# =============================================================================
# BEND DEDUCTION
# =============================================================================

def bend_deduction(
    angulo: float,
    raio: float,
    espessura: float,
    k: float,
) -> float:

    theta_meio = math.radians(
        abs(angulo) / 2.0
    )

    ba = bend_allowance(
        angulo,
        raio,
        espessura,
        k,
    )

    return (
        2.0 *
        (
            raio +
            espessura
        )
        *
        math.tan(
            theta_meio
        )
        -
        ba
    )


# =============================================================================
# SETBACK
# =============================================================================

def bend_setback(
    angulo: float,
    raio: float,
    espessura: float,
) -> float:

    return (
        (
            raio +
            espessura
        )
        *
        math.tan(
            math.radians(
                abs(angulo) / 2.0
            )
        )
    )


# =============================================================================
# DESENVOLVIMENTO
# =============================================================================

def estimar_desenvolvimento(
    contorno: Contorno,
    dobras: List[Dobra],
    espessura_mm: Optional[float],
    deteccao_geometrica: dict,
) -> Tuple[float, dict]:
    """
    Calcula o desenvolvimento real no eixo transversal da pe?a.

    REGRA:

        ABA 1
        + BA 1
        + ABA 2
        + BA 2
        + ABA 3

    IMPORTANTE:

    A dimens?o longitudinal da pe?a N?O participa do desenvolvimento.

    Exemplo do componente 046:

        longitudinal ~= 672 mm
        transversal ~= 163 mm

    O blank final ser?:

        largura  = desenvolvimento
        comprimento = 672 mm

    Para duas dobras de 90 graus:

        BA = theta * (R + K*T)

        BD = 2*(R+T)*tan(theta/2) - BA

    As abas s?o obtidas geometricamente a partir das posi??es
    reais das linhas de dobra.
    """

    if espessura_mm is None:
        espessura_mm = ESPESSURA_PADRAO

    if len(dobras) != 2:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INSUFICIENTE",
                "motivo": "QUANTIDADE_DE_DOBRAS_DIFERENTE_DE_2",
                "base_mm": 0.0,
                "bend_allowance_total_mm": 0.0,
                "bend_deduction_total_mm": 0.0,
                "desenvolvimento_mm": 0.0,
                "detalhes": [],
                "sequencia": [],
            },
        )

    if any(d.posicao_mm is None for d in dobras):
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INSUFICIENTE",
                "motivo": "POSICOES_REAIS_DAS_DOBRAS_NAO_IDENTIFICADAS",
                "base_mm": 0.0,
                "bend_allowance_total_mm": 0.0,
                "bend_deduction_total_mm": 0.0,
                "desenvolvimento_mm": 0.0,
                "detalhes": [],
                "sequencia": [],
            },
        )

    eixo = deteccao_geometrica.get("eixo_longitudinal")

    if eixo == "X":
        dimensao_transversal_mm = min(contorno.largura_mm, contorno.altura_mm)
        comprimento_longitudinal_mm = contorno.largura_mm
        eixo_desenvolvimento = "TRANSVERSAL_Y"

    elif eixo == "Y":
        dimensao_transversal_mm = min(contorno.largura_mm, contorno.altura_mm)
        comprimento_longitudinal_mm = contorno.altura_mm
        eixo_desenvolvimento = "TRANSVERSAL_X"

    else:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INSUFICIENTE",
                "motivo": "EIXO_DAS_DOBRAS_NAO_IDENTIFICADO",
                "base_mm": 0.0,
                "bend_allowance_total_mm": 0.0,
                "bend_deduction_total_mm": 0.0,
                "desenvolvimento_mm": 0.0,
                "sequencia": [],
            },
        )

    dobras_ordenadas = ordenar_dobras(dobras)

    p1 = dobras_ordenadas[0].posicao_mm
    p2 = dobras_ordenadas[1].posicao_mm

    if p1 is None or p2 is None:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INSUFICIENTE",
                "motivo": "POSICAO_DE_DOBRA_AUSENTE",
                "base_mm": dimensao_transversal_mm,
                "sequencia": [],
            },
        )

    if p2 <= p1:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INCONSISTENTE",
                "motivo": "ORDEM_DAS_DOBRAS_INVALIDA",
                "posicao_1_mm": p1,
                "posicao_2_mm": p2,
                "base_mm": dimensao_transversal_mm,
                "sequencia": [],
            },
        )

    if p1 <= 0 or p2 >= dimensao_transversal_mm:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INCONSISTENTE",
                "motivo": "POSICAO_DOBRA_FORA_DO_CONTORNO",
                "posicao_dobra_1_mm": p1,
                "posicao_dobra_2_mm": p2,
                "dimensao_transversal_mm": dimensao_transversal_mm,
                "sequencia": [],
            },
        )

    # -------------------------------------------------------------------------
    # PERFIL TRANSVERSAL
    # -------------------------------------------------------------------------

    perfil = reconstruir_perfil_transversal(
        contorno,
        dobras_ordenadas,
        espessura_mm,
    )

    # -------------------------------------------------------------------------
    # TRECHOS GEOM?TRICOS
    # -------------------------------------------------------------------------

    trecho_externo_1 = p1
    trecho_entre_dobras = p2 - p1
    trecho_externo_3 = dimensao_transversal_mm - p2

    # -------------------------------------------------------------------------
    # BA / BD / SETBACK
    # -------------------------------------------------------------------------

    detalhes = []
    setbacks = []

    ba_total = 0.0
    bd_total = 0.0

    for dobra in dobras_ordenadas:

        raio = (
            dobra.raio
            if dobra.raio is not None
            else RAIO_PADRAO
        )

        k = fator_k_por_raio(
            raio,
            espessura_mm,
        )

        ba = bend_allowance(
            dobra.angulo,
            raio,
            espessura_mm,
            k,
        )

        bd = bend_deduction(
            dobra.angulo,
            raio,
            espessura_mm,
            k,
        )

        setback = bend_setback(
            dobra.angulo,
            raio,
            espessura_mm,
        )

        ba_total += ba
        bd_total += bd
        setbacks.append(setback)

        detalhes.append(
            {
                "dobra": dobra.numero,
                "angulo_graus": dobra.angulo,
                "raio_mm": raio,
                "espessura_mm": espessura_mm,
                "razao_raio_espessura": (
                    raio / espessura_mm
                    if espessura_mm
                    else None
                ),
                "fator_k": k,
                "setback_mm": setback,
                "bend_allowance_mm": ba,
                "bend_deduction_mm": bd,
                "posicao_mm": dobra.posicao_mm,
                "origem_posicao": dobra.origem,
            }
        )

    # -------------------------------------------------------------------------
    # ABAS REAIS
    # -------------------------------------------------------------------------

    aba_1 = trecho_externo_1 - setbacks[0]

    aba_2 = (
        trecho_entre_dobras
        - setbacks[0]
        - setbacks[1]
    )

    aba_3 = trecho_externo_3 - setbacks[1]

    abas_validas = (
        aba_1 > 0
        and aba_2 > 0
        and aba_3 > 0
    )

    if not abas_validas:
        return (
            0.0,
            {
                "status": "NAO_CALCULADO",
                "metodo": "GEOMETRIA_INCONSISTENTE",
                "motivo": "UMA_DAS_ABAS_E_NEGATIVA_OU_ZERO",
                "aba_1_mm": aba_1,
                "aba_2_mm": aba_2,
                "aba_3_mm": aba_3,
                "perfil_transversal": perfil,
                "detalhes": detalhes,
                "sequencia": [],
            },
        )

    # -------------------------------------------------------------------------
    # DESENVOLVIMENTO REAL
    #
    # ABA1 + BA1 + ABA2 + BA2 + ABA3
    #
    # NUNCA usar bounding box longitudinal aqui.
    # -------------------------------------------------------------------------

    ba_1 = detalhes[0]["bend_allowance_mm"]
    ba_2 = detalhes[1]["bend_allowance_mm"]

    desenvolvimento = (
        aba_1
        + ba_1
        + aba_2
        + ba_2
        + aba_3
    )

    # M?todo alternativo apenas para confer?ncia.
    desenvolvimento_por_bd = (
        dimensao_transversal_mm
        - bd_total
    )

    diferenca_metodos = (
        desenvolvimento
        - desenvolvimento_por_bd
    )

    # -------------------------------------------------------------------------
    # SEQU?NCIA EXPL?CITA
    # -------------------------------------------------------------------------

    sequencia = [
        {
            "tipo": "ABA",
            "numero": 1,
            "comprimento_mm": aba_1,
            "trecho_externo_mm": trecho_externo_1,
            "setback_mm": setbacks[0],
            "origem": "GEOMETRIA_ABA_1",
        },
        {
            "tipo": "BA",
            "dobra": 1,
            "comprimento_mm": ba_1,
            "angulo_graus": detalhes[0]["angulo_graus"],
            "raio_mm": detalhes[0]["raio_mm"],
            "fator_k": detalhes[0]["fator_k"],
        },
        {
            "tipo": "ABA",
            "numero": 2,
            "comprimento_mm": aba_2,
            "trecho_externo_mm": trecho_entre_dobras,
            "setback_1_mm": setbacks[0],
            "setback_2_mm": setbacks[1],
            "origem": "GEOMETRIA_ABA_2",
        },
        {
            "tipo": "BA",
            "dobra": 2,
            "comprimento_mm": ba_2,
            "angulo_graus": detalhes[1]["angulo_graus"],
            "raio_mm": detalhes[1]["raio_mm"],
            "fator_k": detalhes[1]["fator_k"],
        },
        {
            "tipo": "ABA",
            "numero": 3,
            "comprimento_mm": aba_3,
            "trecho_externo_mm": trecho_externo_3,
            "setback_mm": setbacks[1],
            "origem": "GEOMETRIA_ABA_3",
        },
    ]

    soma_sequencia = sum(
        item["comprimento_mm"]
        for item in sequencia
    )

    erro_fechamento = (
        desenvolvimento
        - soma_sequencia
    )

    deteccao_confirmada = (
        deteccao_geometrica.get("encontradas", False)
        and deteccao_geometrica.get("quantidade", 0) == 2
    )

    perfil_confirmado = (
        perfil.get("status")
        in ("CONFIRMADO", "PARCIAL")
    )

    perfil_nao_inconsistente = (
        perfil.get("status")
        != "GEOMETRIA_INCONSISTENTE"
    )

    valido = (
        desenvolvimento > 0
        and abs(erro_fechamento) <= TOLERANCIA_RECONSTRUCAO_MM
        and deteccao_confirmada
        and perfil_nao_inconsistente
    )

    return (
        desenvolvimento,
        {
            "status": (
                "CALCULADO"
                if valido
                else "NAO_CALCULADO"
            ),

            "metodo":
                "GEOMETRIA_ABA_BA_ABA_BA_ABA",

            "eixo_desenvolvimento":
                eixo_desenvolvimento,

            "eixo_longitudinal":
                eixo,

            "comprimento_longitudinal_mm":
                comprimento_longitudinal_mm,

            "base_mm":
                dimensao_transversal_mm,

            "dimensao_transversal_mm":
                dimensao_transversal_mm,

            "posicao_dobra_1_mm":
                p1,

            "posicao_dobra_2_mm":
                p2,

            "trecho_externo_1_mm":
                trecho_externo_1,

            "trecho_entre_dobras_mm":
                trecho_entre_dobras,

            "trecho_externo_3_mm":
                trecho_externo_3,

            "setback_1_mm":
                setbacks[0],

            "setback_2_mm":
                setbacks[1],

            "aba_1_mm":
                aba_1,

            "aba_2_mm":
                aba_2,

            "aba_3_mm":
                aba_3,

            "bend_allowance_total_mm":
                ba_total,

            "bend_deduction_total_mm":
                bd_total,

            "desenvolvimento_mm":
                desenvolvimento,

            "desenvolvimento_por_bd_mm":
                desenvolvimento_por_bd,

            "diferenca_entre_metodos_mm":
                diferenca_metodos,

            "soma_sequencia_mm":
                soma_sequencia,

            "erro_fechamento_mm":
                erro_fechamento,

            "detalhes":
                detalhes,

            "sequencia":
                sequencia,

            "perfil_transversal":
                perfil,

            "deteccao_geometrica":
                deteccao_geometrica,

            "perfil_geometrico_confirmado":
                perfil_confirmado,

            "criterio":
                "SEGMENTOS_LONGITUDINAIS_REAIS_DOBRAS_GEOMETRICAS_ABA_BA_ABA_BA_ABA",

            "deteccao_dobras_confirmada":
                deteccao_confirmada,

            "nao_utiliza_bounding_box_para_desenvolvimento":
                True,
        },
    )

# =============================================================================
# BLANK
# =============================================================================

def calcular_blank(
    contorno: Contorno,
    desenvolvimento_mm: float,
    espessura_mm: Optional[float],
    desenvolvimento_detalhes: dict,
) -> Blank:

    status = (
        desenvolvimento_detalhes.get(
            "status",
            "NAO_CALCULADO",
        )
    )

    if (
        status != "CALCULADO"
        or
        desenvolvimento_mm <= 0
    ):

        return Blank(
            comprimento_desenvolvido_mm=0.0,
            largura_blank_mm=0.0,
            comprimento_blank_mm=0.0,
            espessura_mm=espessura_mm,
            metodo=(
                "NAO_CALCULADO_GEOMETRIA_INSUFICIENTE"
            ),
            valido=False,
        )

    # =====================================================================
    # COMPRIMENTO LONGITUDINAL DO BLANK
    #
    # O eixo X/Y do PDF NÃO determina a orientação física do blank.
    #
    # O desenvolvimento já representa a dimensão transversal desenvolvida.
    # A dimensão longitudinal original da peça é a maior dimensão do
    # contorno físico.
    #
    # Componente 046:
    #
    #   Contorno = 672.0023 x 162.9995 mm
    #
    #   Desenvolvimento = 136.5960 mm
    #   Longitudinal   = 672.0023 mm
    #
    # Portanto:
    #
    #   Blank = 136.5960 x 672.0023 mm
    #
    # NÃO utilizar aqui:
    #
    #   eixo_longitudinal
    #   eixo PDF
    #   desenvolvimento_detalhes["comprimento_longitudinal_mm"]
    #
    # porque esses dados são da detecção geométrica do desenho e podem
    # representar a orientação dos eixos PDF, não a orientação física
    # final do blank.
    # =====================================================================

    comprimento_longitudinal = max(
        contorno.largura_mm,
        contorno.altura_mm,
    )

    if comprimento_longitudinal <= 0:
        return Blank(
            comprimento_desenvolvido_mm=0.0,
            largura_blank_mm=0.0,
            comprimento_blank_mm=0.0,
            espessura_mm=espessura_mm,
            metodo="NAO_CALCULADO_DIMENSAO_LONGITUDINAL_INVALIDA",
            valido=False,
        )

    return Blank(
        comprimento_desenvolvido_mm=
            desenvolvimento_mm,

        largura_blank_mm=
            desenvolvimento_mm,

        comprimento_blank_mm=
            comprimento_longitudinal,

        espessura_mm=
            espessura_mm,

        metodo=
            "DESENVOLVIMENTO_GEOMETRICO_ABAS_BA",

        valido=True,
    )

# =============================================================================
# DETECTAR CONTORNOS
# =============================================================================

def detectar_contornos(
    segmentos: List[Segmento],
) -> List[Contorno]:

    grupos = agrupar_segmentos_conectados(
        segmentos
    )

    contornos = []

    for i, grupo in enumerate(
        grupos
    ):

        if len(grupo) < 3:
            continue

        contorno = criar_contorno(
            grupo,
            f"{i + 1:03d}",
        )

        if contorno is None:
            continue

        contornos.append(
            contorno
        )

    contornos.sort(
        key=lambda c:
            c.area_pdf,
        reverse=True,
    )

    for i, contorno in enumerate(
        contornos,
        start=1,
    ):

        contorno.componente = (
            f"{i:03d}"
        )

    return contornos


# =============================================================================
# SELECIONAR COMPONENTE 046
# =============================================================================

def selecionar_046(
    contornos: List[Contorno],
) -> Optional[Contorno]:

    melhores = []

    for contorno in contornos:

        if (
            contorno.largura_pdf <= 0
            or
            contorno.altura_pdf <= 0
        ):
            continue

        pontuar_contorno(
            contorno,
            DIMENSAO_ALVO_LARGURA,
            DIMENSAO_ALVO_ALTURA,
        )

        melhores.append(
            contorno
        )

    if not melhores:
        return None

    alvo_proporcao = (
        DIMENSAO_ALVO_LARGURA /
        DIMENSAO_ALVO_ALTURA
    )

    def erro_proporcao(c):

        proporcao = (
            max(
                c.largura_pdf,
                c.altura_pdf,
            )
            /
            max(
                min(
                    c.largura_pdf,
                    c.altura_pdf,
                ),
                0.0001,
            )
        )

        return abs(
            proporcao -
            alvo_proporcao
        )

    melhores.sort(
        key=lambda c: (
            -c.score,
            erro_proporcao(c),
        )
    )

    candidato = melhores[0]

    proporcao = (
        max(
            candidato.largura_pdf,
            candidato.altura_pdf,
        )
        /
        max(
            min(
                candidato.largura_pdf,
                candidato.altura_pdf,
            ),
            0.0001,
        )
    )

    if proporcao < 2.5:

        alongados = [
            c
            for c in melhores
            if (
                max(
                    c.largura_pdf,
                    c.altura_pdf,
                )
                /
                max(
                    min(
                        c.largura_pdf,
                        c.altura_pdf,
                    ),
                    0.0001,
                )
                >= 2.5
            )
        ]

        if alongados:

            candidato = alongados[0]

    candidato.componente = "046"

    return candidato


# =============================================================================
# CLASSIFICAR SEGMENTO
# =============================================================================

def classificar_segmento(
    segmento: Segmento,
) -> str:

    comprimento = (
        segmento.comprimento
    )

    if comprimento < 0.5:
        return "AUXILIAR"

    angulo = (
        abs(
            segmento.angulo
        )
        %
        180
    )

    if (
        abs(angulo) < 1
        or
        abs(angulo - 180) < 1
        or
        abs(angulo - 90) < 1
    ):

        return "RETO"

    return "RETO_INCLINADO"


# =============================================================================
# SEGMENTOS EM MM
# =============================================================================

def segmentos_em_mm(
    contorno: Contorno,
) -> List[dict]:

    resultado = []

    for i, s in enumerate(
        contorno.segmentos,
        start=1,
    ):

        resultado.append(
            {
                "segmento":
                    i,

                "x1_mm":
                    round(
                        s.x1 *
                        contorno.escala,
                        4,
                    ),

                "y1_mm":
                    round(
                        s.y1 *
                        contorno.escala,
                        4,
                    ),

                "x2_mm":
                    round(
                        s.x2 *
                        contorno.escala,
                        4,
                    ),

                "y2_mm":
                    round(
                        s.y2 *
                        contorno.escala,
                        4,
                    ),

                "comprimento_mm":
                    round(
                        s.comprimento *
                        contorno.escala,
                        4,
                    ),

                "angulo_graus":
                    round(
                        s.angulo,
                        4,
                    ),

                "tipo":
                    classificar_segmento(
                        s
                    ),
            }
        )

    return resultado

# PROCESSAMENTO PRINCIPAL
# =============================================================================

def processar(
    caminho_pdf: str,
):

    caminho = Path(
        caminho_pdf
    )

    if not caminho.exists():

        raise FileNotFoundError(
            f"Arquivo nÃ£o encontrado: "
            f"{caminho}"
        )

    print("=" * 80)
    print(
        "AIZI ENGINEERING AI"
    )
    print(
        "DIAGNÃ“STICO DE DESENHO"
    )
    print("=" * 80)

    print(
        f"Arquivo: {caminho}"
    )

    documento = pymupdf.open(
        caminho
    )

    todos_segmentos = []
    todos_textos = []

    for numero_pagina, pagina in enumerate(
        documento,
        start=1,
    ):

        print(
            f"Processando pÃ¡gina "
            f"{numero_pagina}..."
        )

        segmentos = extrair_linhas(
            pagina
        )

        textos = extrair_textos(
            pagina
        )

        todos_segmentos.extend(
            segmentos
        )

        todos_textos.extend(
            textos
        )

    documento.close()

    print(
        f"Linhas vetoriais encontradas: "
        f"{len(todos_segmentos)}"
    )

    print(
        f"Textos encontrados: "
        f"{len(todos_textos)}"
    )

    # -------------------------------------------------------------------------
    # CONTORNOS
    # -------------------------------------------------------------------------

    contornos = detectar_contornos(
        todos_segmentos
    )

    print(
        f"Contornos encontrados: "
        f"{len(contornos)}"
    )

    if not contornos:

        raise RuntimeError(
            "Nenhum contorno fechado encontrado."
        )

    # -------------------------------------------------------------------------
    # COMPONENTE 046
    # -------------------------------------------------------------------------

    principal = selecionar_046(
        contornos
    )

    if principal is None:

        raise RuntimeError(
            "NÃ£o foi possÃ­vel identificar "
            "o contorno principal."
        )

    principal.componente = "046"

    # -------------------------------------------------------------------------
    # ESPESSURA
    # -------------------------------------------------------------------------

    espessura = identificar_espessura(
        todos_textos
    )

    if espessura is None:

        espessura = ESPESSURA_PADRAO

    # -------------------------------------------------------------------------
    # DOBRAS
    # -------------------------------------------------------------------------

    dobras, deteccao_geometrica = (
        identificar_dobras(
            todos_textos,
            principal,
            espessura,
            todos_segmentos,
        )
    )

    dobras = ordenar_dobras(
        dobras
    )

    # -------------------------------------------------------------------------
    # DESENVOLVIMENTO
    # -------------------------------------------------------------------------

    desenvolvimento, detalhes = (
        estimar_desenvolvimento(
            principal,
            dobras,
            espessura,
            deteccao_geometrica,
        )
    )

    # -------------------------------------------------------------------------
    # BLANK
    # -------------------------------------------------------------------------

    blank = calcular_blank(
        principal,
        desenvolvimento,
        espessura,
        detalhes,
    )

    # -------------------------------------------------------------------------
    # PASTA DE SAÃDA
    # -------------------------------------------------------------------------

    pasta_saida = (
        caminho.parent /
        caminho.stem
    )

    pasta_saida.mkdir(
        parents=True,
        exist_ok=True,
    )

    caminho_txt = (
        pasta_saida /
        "_diagnostico.txt"
    )

    caminho_json = (
        pasta_saida /
        "_calculadora_corte.json"
    )

    # -------------------------------------------------------------------------
    # TXT
    # -------------------------------------------------------------------------

    gerar_relatorio(
        caminho,
        caminho_txt,
        principal,
        dobras,
        blank,
        espessura,
        detalhes,
        todos_textos,
    )

    # -------------------------------------------------------------------------
    # JSON
    # -------------------------------------------------------------------------

    gerar_json_calculadora_corte(
        caminho_json,
        principal,
        dobras,
        blank,
        espessura,
        detalhes,
    )

    # -------------------------------------------------------------------------
    # RESULTADO
    # -------------------------------------------------------------------------

    imprimir_resultado(
        principal,
        dobras,
        blank,
        caminho_txt,
        caminho_json,
        detalhes,
    )

    return {

        "componente":
            "046",

        "largura_mm":
            principal.largura_mm,

        "altura_mm":
            principal.altura_mm,

        "area_mm2":
            principal.area_mm2,

        "perimetro_mm":
            principal.perimetro_mm,

        "segmentos":
            len(
                principal.segmentos
            ),

        "dobras":
            [
                asdict(d)
                for d in dobras
            ],

        "deteccao_geometrica":
            deteccao_geometrica,

        "desenvolvimento_mm":
            blank.comprimento_desenvolvido_mm,

        "desenvolvimento_detalhes":
            detalhes,

        "blank": {

            "largura_mm":
                blank.largura_blank_mm,

            "comprimento_mm":
                blank.comprimento_blank_mm,

            "espessura_mm":
                blank.espessura_mm,

            "valido":
                blank.valido,
        },

        "diagnostico":
            str(caminho_txt),

        "calculadora_corte":
            str(caminho_json),
    }
